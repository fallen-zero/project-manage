```rust
    // 减锁（M5.5 D6）：`index_text` = `cut_for_search` + 逐词 String 分配 + join，是**纯 CPU**。
    // 原先它在下面 `params![..]` 里求值，而事务第一句 INSERT（本函数 :49 那笔）早已拿到
    // SQLite 写锁 ⇒ 锁持有时间 ≈ 分词一整篇正文 + 两条 INSERT，自动轮进行中别的 sync 命令
    // 就在 `busy_timeout` 上排队（M5 实测 4.2 s / 一次 5565 ms db_failed 的直接来源）。
    // 提到开事务之前，锁里只剩两条 INSERT（微秒级）。抽取本身本就在锁外（run_pass 的
    // `match panic_to_err(|| extract_text(..))` 先抽、其 Ok arm 才调本函数写库）。
    let name_tokens = index_text(&file.file_name);
    let body_tokens = body.as_ref().map(|b| index_text(b));

    let tx = conn.unchecked_transaction()?;
    // …（下面 INSERT index_docs 那笔、query_row 取 rowid、DELETE fts 一字不改）…
    tx.execute("DELETE FROM index_docs_fts WHERE rowid = ?1", params![rowid])?;
    if let Some(tokens) = body_tokens {
        tx.execute(
            "INSERT INTO index_docs_fts (rowid, name_tokens, body_tokens) VALUES (?1, ?2, ?3)",
            params![rowid, name_tokens, tokens],
        )?;
    }
    tx.commit()?;
```

注意：`name_tokens` 只在有 `body` 时才用得上，但 `index_text(&file.file_name)` 很便宜且无条件算一次不影响正确性；若想更省可把 `name_tokens` 也塞进 `if let`。**保持 `body` 的 `Option` 判定与原 `if let Some(body) = body` 完全一致**（原本 `DocOutcome::Ok` 且 trim 非空才 `Some`），不要改这层语义。

```rust
//! 文件监听（M5.5 唯一新机制）。这一层**只置一个脏布尔**，绝不碰 SQLite：
//! notify 回调跑在 notify 内部线程，若在那儿直接写库就绕过了 index_job 的 running 单飞闸门。
//! 真正的追平仍由 ticker 调既有 index_job::start(None, rebuild=false) 走那一整轮（spec D1/D3）。

use std::sync::atomic::{AtomicBool, Ordering};
use std::sync::Arc;

/// 「自上次追平之后有没有过文件事件」的唯一位点。D2：不是项目集合，就是一个布尔
/// （index_job::start 只吃单个 Option<String> project_id、targets() 追平轮本就该扫全部）。
pub struct WatchShared {
    pub dirty: AtomicBool,
    /// 与 IndexShared.running 共享同一个 Arc，ticker 用它判「有没有轮在跑」。
    pub running: Arc<AtomicBool>,
}

impl WatchShared {
    pub fn new(running: Arc<AtomicBool>) -> Self {
        Self { dirty: AtomicBool::new(false), running }
    }
}

pub fn mark_dirty(shared: &WatchShared) {
    shared.dirty.store(true, Ordering::SeqCst);
}

/// ticker 的决策：先 swap 清脏、再看 running。顺序即判据（spec 第四节①）——
/// 「先起轮后清脏」会让轮跑期间进来的新事件被下一轮重复起。用一次 swap 把读+清并成原子。
pub fn due_for_pass(shared: &WatchShared) -> bool {
    let was_dirty = shared.dirty.swap(false, Ordering::SeqCst);
    was_dirty && !shared.running.load(Ordering::SeqCst)
}
```

```rust
/// 起 notify watcher：监听所有「未软删且有 root」的项目目录，事件回调只做 `mark_dirty`。
/// 注册失败（root 不可达是常态用法）跳过该 root、继续其余、不写 last_error、不阻断启动（spec 第六节）。
/// 返回注册不成功的根路径列表，供上层记一句日志（本批只回不给界面）。
pub fn spawn_watcher(data_dir: &std::path::Path, shared: Arc<WatchShared>) -> AppResult<Vec<String>> {
    use notify::{Config, Event, RecommendedWatcher, RecursiveMode, Watcher};
    let conn = db::open(data_dir)?; // 只读列 root
    let mut roots = list_watchable_roots(&conn)?; // Vec<String>：未软删项目的 root 路径
    // 过滤掉当前不可达的（is_dir 为假的不注册，避免噪声）
    roots.retain(|p| std::path::Path::new(p).is_dir());
    let mut failed = Vec::new();
    let mut watcher = RecommendedWatcher::new(
        move |res: Result<Event, notify::Error>| {
            if res.is_ok() {
                mark_dirty(&shared); // ← 唯一副作用：置脏。绝不在回调里写库。
            }
        },
        Config::default(),
    )
    .map_err(|e| AppError::new("watch_failed", &format!("文件监听初始化失败：{e}"), None))?;
    for r in roots {
        if watcher.watch(std::path::Path::new(&r), RecursiveMode::Recursive).is_err() {
            failed.push(r);
        }
    }
    // watcher 必须活过整个进程 —— 交给一个 std::thread 持有并阻塞（不 mem::forget，让线程句柄托管）。
    std::thread::Builder::new().name("file-watch".into()).spawn(move || {
        let _keep = watcher; // 线程存活期间 watcher 不被 drop
        loop { std::thread::park(); } // 永阻塞；进程退出即回收（风险 2：常驻句柄生命周期）
    })?;
    Ok(failed)
}

fn list_watchable_roots(conn: &rusqlite::Connection) -> AppResult<Vec<String>> {
    let mut stmt = conn.prepare(
        "SELECT d.path FROM project_dirs d JOIN projects p ON p.id = d.project_id
          WHERE d.kind = 'root' AND p.deleted_at IS NULL",
    )?;
    let rows = stmt.query_map([], |r| r.get::<_, String>(0))?
        .collect::<Result<Vec<_>, _>>()?;
    Ok(rows)
}
```

pub fn spawn_ticker(

    shared: Arc<WatchShared>,

```rust
/// 前缀码元长度：prefix[i] = chars[..i] 的 UTF-16 码元数。把 stitch 里「每命中一次 utf16_len」
/// 的 O(命中数×窗长) 摊成 O(n)（M5 真机：8MB×19.5万命中曾致单次预览挂起 >9 分钟）。
fn utf16_prefix(chars: &[char]) -> Vec<usize> {
    let mut p = Vec::with_capacity(chars.len() + 1);
    p.push(0usize);
    for c in chars {
        let last = *p.last().unwrap();
        p.push(last + c.len_utf16());
    }
    p
}
```
